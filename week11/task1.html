function ProjectCard({ title, description, tech, github, demo }) {
  return (
    <article className="project-card">
      <h3 className="project-title">{title}</h3>
      <p className="project-desc">{description}</p>
      <ul className="tech-list" aria-label="Tech stack">
        {tech.map((t) => <li key={t} className="tech-tag">{t}</li>)}
      </ul>
      <div className="project-links">
        <a href={github} target="_blank" rel="noopener noreferrer">GitHub ↗</a>
        <a href={demo} target="_blank" rel="noopener noreferrer">Live demo ↗</a>
      </div>
    </article>
  );
}

export default ProjectCard;

ProjectList — 6 cards from an array

src/components/ProjectList.jsx

Maps the projects array to cards, each with a stable key.

import ProjectCard from './ProjectCard';

function ProjectList({ projects }) {
  if (projects.length === 0) {
    return <p className="empty">No projects match this filter.</p>;
  }

  return (
    <div className="project-grid">
      {projects.map((project) => (
        <ProjectCard
          key={project.id}
          title={project.title}
          description={project.description}
          tech={project.tech}
          github={project.github}
          demo={project.demo}
        />
      ))}
    </div>
  );
}

export default ProjectList;

FilterBar and state in App

src/App.jsx · src/components/FilterBar.jsx

The active technology lives in App with useState; FilterBar triggers changes and ProjectList receives the filtered array.

import { useState } from 'react';
import projects from './data/projects';
import FilterBar from './components/FilterBar';
import ProjectList from './components/ProjectList';

function App() {
  const [active, setActive] = useState('All');

  const technologies = ['All', ...new Set(projects.flatMap((p) => p.tech))];

  const visible = active === 'All'
    ? projects
    : projects.filter((p) => p.tech.includes(active));

  return (
    <>
      <FilterBar technologies={technologies} active={active} onChange={setActive} />
      <ProjectList projects={visible} />
    </>
  );
}

export default App;
